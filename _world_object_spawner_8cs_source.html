63 for (int i = 0; i <= totalToSpawn;)

80 // We use GridMaxX for both because we want the values as percentages of the max.

92 private Vector3 GetClearPosWithDistance(int min, int max)
93 {
94 bool posFound = false;
95 Vector3 ranPos = Vector3.zero;
96 int breakcount = 0;
97 while (posFound == false)
98 {
99 breakcount++;
100 if (breakcount > 1000)
101 {
102 Debug.Log("Break point in GetClearPosWithDistance hit.");
103 return Vector3.zero;
104 }
105 ranPos = new Vector3(Random.Range(-max, max + 1), 1, Random.Range(-max, max + 1));
106
107 // Check min distances, avoids having to use Vector3.Distance
108 if (ranPos.x <= -min || ranPos.x >= min
109 || ranPos.z <= -min || ranPos.z >= min)
110 {
111 if (grid.IsNodeClear(ranPos)) posFound = true;
112 }
113 }
114
115 return ranPos;
116 }
